Scientific Python Basics

A hands-on introduction to the scientific Python stack, from the language itself to NumPy, Matplotlib, SciPy, pandas and scikit-learn, with examples from chemistry and physics.

import numpy as np
from scipy.integrate import solve_ivp

# A → B → C, first-order kinetics
def rates(t, c, k1=1.0, k2=0.5):
    a, b, _ = c
    return [-k1*a, k1*a - k2*b, k2*b]

t = np.linspace(0, 10, 200)
sol = solve_ivp(rates, (0, 10), [1, 0, 0], t_eval=t)
